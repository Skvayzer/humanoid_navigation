cmake_minimum_required(VERSION 3.10)
project(g1_cat_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(utlidar_to_livox src/utlidar_to_livox.cpp)
target_compile_features(utlidar_to_livox PRIVATE cxx_std_14)
target_compile_options(utlidar_to_livox PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(utlidar_to_livox rclcpp sensor_msgs livox_ros_driver2)
install(TARGETS utlidar_to_livox DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/stationary_pose.py DESTINATION lib/${PROJECT_NAME} RENAME stationary_pose)
ament_package()
